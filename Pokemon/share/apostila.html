Fluxo de integração

Arquivo JSON → objeto de entrada
             → requisições HTTP
             → respostas da API
             → objetos internos
             → arquivo JSON de saída
Não acople o arquivo final ao contrato completo de uma API externa. Mapeie apenas os dados necessários.

Modelos mínimos da PokéAPI

public sealed class PokemonApiResponse
{
    public int Id { get; init; }
    public string Name { get; init; } = "";
    public int Height { get; init; }
    public int Weight { get; init; }
    public PokemonSprites Sprites { get; init; } = new();
    public List<PokemonTypeSlot> Types { get; init; } = [];
}

public sealed class PokemonSprites
{
    [JsonPropertyName("front_default")]
    public string? FrontDefault { get; init; }
}

public sealed class PokemonTypeSlot
{
    public PokemonType Type { get; init; } = new();
}

public sealed class PokemonType
{
    public string Name { get; init; } = "";
}
